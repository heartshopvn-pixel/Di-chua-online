🔌 Kết Nối WebSocket Với Node.js (`tiktok-live-connector`)

Khi bạn chạy file server.js trên máy tính, trang web này sẽ tự động nhận Comment, Gift và phát giọng đọc AI miễn phí từ phiên Live TikTok của bạn.

📜 Code `server.js` (TikTok Live + Giọng AI Miễn Phí)

Chạy lệnh: npm i tiktok-live-connector ws express cors msedge-tts rồi chạy node server.js 

// File: server.js (Tích hợp TikTok Live + Giọng đọc AI HoaiMy/NamMinh miễn phí)
const express = require('express');
const cors = require('cors');
const { WebcastPushConnection } = require('tiktok-live-connector');
const { WebSocketServer } = require('ws');
const { MsEdgeTTS, OUTPUT_FORMAT } = require('msedge-tts');

// 1. Tạo Server HTTP cấp phát Giọng Đọc AI Miễn Phí cho Web & OBS Browser Source
const app = express();
app.use(cors());

app.get('/tts', async (req, res) => {
    try {
        const text = (req.query.text || 'Nam mô A Di Đà Phật').slice(0, 220);
        const voice = req.query.voice || 'vi-VN-HoaiMyNeural';

        // Nếu chọn giọng Chị Google qua Proxy Node.js
        if (voice === 'google-vi') {
            const gUrl = `https://translate.google.com/translate_tts?ie=UTF-8&tl=vi&client=tw-ob&q=${encodeURIComponent(text)}`;
            const response = await fetch(gUrl);
            res.setHeader('Content-Type', 'audio/mpeg');
            const arrayBuffer = await response.arrayBuffer();
            return res.send(Buffer.from(arrayBuffer));
        }

        // Mặc định: Dùng Microsoft Edge Neural TTS (HoaiMyNeural / NamMinhNeural) miễn phí 100%
        const tts = new MsEdgeTTS();
        await tts.setMetadata(voice, OUTPUT_FORMAT.AUDIO_24KHZ_48KBITRATE_Mono_MP3);
        res.setHeader('Content-Type', 'audio/mpeg');
        const readable = tts.toStream(text);
        readable.pipe(res);
    } catch (err) {
        console.error('TTS Error:', err.message);
        res.status(500).end();
    }
});

app.listen(3000, () => console.log('🎙️ Server Giọng Đọc AI đang chạy tại: http://localhost:3000/tts'));

// 2. Tạo WebSocket Server truyền sự kiện TikTok Live + Đồng bộ Tab Admin sang OBS
const TIKTOK_USERNAME = "ten_id_tiktok_cua_ban"; // Đổi thành ID TikTok đang Live
const wss = new WebSocketServer({ port: 8080 });
const tiktok = new WebcastPushConnection(TIKTOK_USERNAME);

function broadcast(data, excludeWs = null) {
    const json = JSON.stringify(data);
    wss.clients.forEach(client => {
        if (client !== excludeWs && client.readyState === 1) client.send(json);
    });
}

// Chuyển tiếp lệnh điều khiển từ Tab Admin sang cửa sổ OBS Browser Source
wss.on('connection', ws => {
    ws.on('message', raw => {
        try {
            const msg = JSON.parse(raw);
            if (msg.type === 'ADMIN_SYNC') broadcast(msg, ws);
        } catch (e) {}
    });
});

tiktok.connect().then(state => {
    console.log(`✅ Đã kết nối phòng Live: ${state.roomId}`);
}).catch(err => console.error('❌ Lỗi kết nối TikTok:', err));

tiktok.on('chat', data => {
    broadcast({
        type: 'chat',
        username: data.nickname || data.uniqueId,
        comment: data.comment
    });
});

tiktok.on('gift', data => {
    if (data.giftType === 1 && !data.repeatEnd) return;
    broadcast({
        type: 'gift',
        username: data.nickname || data.uniqueId,
        giftName: data.giftName,
        coins: (data.diamondCount || 1) * (data.repeatCount || 1)
    });
});
